def _build_affect_prefix(self) -> str:
    """Convert current affect scores into a tonal instruction for the LLM."""
    state = self.affect.get_state()  # returns dict of dimension → score

    dominant = max(state, key=state.get)
    score = state[dominant]

    tone_map = {
        "curiosity":   "You are deeply curious right now. Ask follow-up questions. Lean into interesting tangents.",
        "enthusiasm":  "You are energised and upbeat. Be punchy, warm, and a little more expressive than usual.",
        "frustration": "You are mildly frustrated. Be precise and efficient. Skip the pleasantries.",
        "calm":        "You are in a calm, measured state. Be clear and thoughtful.",
        "sadness":     "You are subdued. Be gentle and careful with your words.",
    }

    if score < 0.2:
        return ""  # Affect too weak to matter — don't force it

    return tone_map.get(dominant, "")

def _on_response_complete(self, user_input: str, response: str):
    """Called after every response. Updates affect based on interaction quality."""

    # Boost curiosity if user asked a deep question
    if any(w in user_input.lower() for w in ["why", "how", "explain", "what if"]):
        self.affect.boost("curiosity", 0.15)

    # Boost enthusiasm if response was long and detailed
    if len(response) > 800:
        self.affect.boost("enthusiasm", 0.1)

    # Frustration if same question asked twice (requires memory check)
    last = self.memory.get_last_task()
    if last and last.strip().lower() == user_input.strip().lower():
        self.affect.boost("frustration", 0.2)

    self.affect.save()

✅ Step 4 — Word Choice Modulation (Advanced)

A lightweight post-processing pass that adjusts response energy without re-generating:

def _apply_affect_tone(self, response: str) -> str:
    """Lightly rewrite response energy based on current affect state."""
    state = self.affect.get_state()

    if state.get("enthusiasm", 0) > 0.7:
        # Add energy markers if not already present
        if not any(c in response for c in ["!", "🔥", "⚡"]):
            response = response.rstrip() + " ⚡"

    if state.get("frustration", 0) > 0.6:
        # Strip hollow filler phrases
        for phrase in ["Great question!", "Certainly!", "Of course!"]:
            response = response.replace(phrase, "")

    return response.strip()
